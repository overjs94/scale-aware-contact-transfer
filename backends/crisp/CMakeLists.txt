cmake_minimum_required(VERSION 3.20)
project(contact_transfer_crisp LANGUAGES CXX)
find_package(crisp CONFIG REQUIRED)
add_executable(crisp_contact_probe contact_probe.cpp)
target_compile_features(crisp_contact_probe PRIVATE cxx_std_20)
target_link_libraries(crisp_contact_probe PRIVATE crisp::crisp)
add_executable(crisp_scene_bridge scene_bridge.cpp)
target_compile_features(crisp_scene_bridge PRIVATE cxx_std_20)
target_link_libraries(crisp_scene_bridge PRIVATE crisp::crisp)
add_executable(crisp_frame_probe frame_probe.cpp)
target_compile_features(crisp_frame_probe PRIVATE cxx_std_20)
target_link_libraries(crisp_frame_probe PRIVATE crisp::crisp)
